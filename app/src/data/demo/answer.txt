(function(){
  'use strict';
  var $ = function(s, r){ return (r || document).querySelector(s); };
  var $$ = function(s, r){ return Array.prototype.slice.call((r || document).querySelectorAll(s)); };
  var reduce = window.matchMedia && matchMedia('(prefers-reduced-motion: reduce)').matches;

  var store = {
    get: function(k, d){ try{ var v = localStorage.getItem('aiclass:' + k); return v == null ? d : JSON.parse(v); }catch(e){ return d; } },
    set: function(k, v){ try{ localStorage.setItem('aiclass:' + k, JSON.stringify(v)); }catch(e){} }
  };

  /* ── toast ── */
  var toastEl = $('#toast'), toastT;
  function toast(msg){
    if(!toastEl) return;
    toastEl.textContent = msg;
    toastEl.classList.add('show');
    clearTimeout(toastT);
    toastT = setTimeout(function(){ toastEl.classList.remove('show'); }, 1700);
  }

  /* ── copy ── */
  function fallbackCopy(t){
    var ta = document.createElement('textarea');
    ta.value = t; ta.setAttribute('readonly', '');
    ta.style.position = 'fixed'; ta.style.top = '-1000px'; ta.style.opacity = '0';
    document.body.appendChild(ta); ta.select();
    var ok = false;
    try{ ok = document.execCommand('copy'); }catch(e){}
    document.body.removeChild(ta);
    return ok;
  }
  function copyText(t){
    if(navigator.clipboard && window.isSecureContext){
      return navigator.clipboard.writeText(t).then(function(){ return true; }, function(){ return fallbackCopy(t); });
    }
    return Promise.resolve(fallbackCopy(t));
  }
  function flash(btn, ok, okText){
    if(!btn.dataset.label) btn.dataset.label = btn.textContent;
    btn.textContent = ok ? (okText || '복사했어요') : '직접 선택해 복사해 주세요';
    btn.classList.add('done');
    clearTimeout(btn._t);
    btn._t = setTimeout(function(){ btn.textContent = btn.dataset.label; btn.classList.remove('done'); }, 1600);
  }

  document.addEventListener('click', function(e){
    var b = e.target.closest ? e.target.closest('[data-copy]') : null;
    if(!b) return;
    var src = b.getAttribute('data-copy'), t = src;
    if(src.charAt(0) === '#'){ var el = $(src); t = el ? el.innerText.trim() : ''; }
    if(!t) return;
    copyText(t).then(function(ok){ flash(b, ok); if(ok) toast('복사했어요 — 붙여 넣으면 돼요'); });
  });

  /* ── schedule / status ── */
  var T0 = Date.parse('2026-11-20T14:00:00+09:00');
  var T1 = Date.parse('2026-11-20T17:00:00+09:00');
  var DUE = Date.parse('2026-11-28T00:00:00+09:00');
  var agenda = [
    [0, 20, '여는 이야기'],
    [20, 70, '실습 1 · 수업 지도안 초안 만들기'],
    [70, 80, '쉬는 시간'],
    [80, 140, '실습 2 · 활동지와 평가 문항 만들기'],
    [140, 170, '실습 3 · 학급 안내문 다듬기'],
    [170, 180, '나눔과 마무리']
  ];
  var statusText = $('#status .status-text');
  var nowEl = $('#now');
  var segs = $$('#ruler .seg');

  function tick(){
    var now = Date.now(), phase;
    segs.forEach(function(s){ s.classList.remove('live', 'past'); });
    if(now < T0){
      phase = 'before';
      statusText.textContent = '11월 20일(금) 14:00에 시작해요';
      if(nowEl) nowEl.hidden = true;
    } else if(now < T1){
      phase = 'live';
      var m = (now - T0) / 60000, cur = agenda[0];
      for(var i = 0; i < agenda.length; i++){ if(m >= agenda[i][0] && m < agenda[i][1]){ cur = agenda[i]; break; } }
      var left = Math.max(1, Math.ceil(cur[1] - m));
      statusText.innerHTML = '';
      statusText.appendChild(document.createTextNode('지금 '));
      var bEl = document.createElement('b'); bEl.textContent = cur[2];
      statusText.appendChild(bEl);
      statusText.appendChild(document.createTextNode(' · ' + left + '분 남았어요'));
      segs.forEach(function(s){
        var f = +s.dataset.from, t = +s.dataset.to;
        if(m >= f && m < t) s.classList.add('live');
        else if(m >= t) s.classList.add('past');
      });
      if(nowEl){ nowEl.hidden = false; nowEl.style.setProperty('--p', (m / 180 * 100).toFixed(2) + '%'); }
    } else if(now < DUE){
      phase = 'after';
      statusText.innerHTML = '연수가 끝났어요 · 선택 과제는 <b>11월 27일(금)</b>까지 패들렛에';
      if(nowEl) nowEl.hidden = true;
    } else {
      phase = 'done';
      statusText.textContent = '연수 자료는 이곳에서 언제든 다시 볼 수 있어요';
      if(nowEl) nowEl.hidden = true;
    }
    document.body.setAttribute('data-phase', phase);
  }
  if(statusText){ tick(); setInterval(tick, 30000); }

  /* ── checklists ── */
  $$('[data-store]').forEach(function(list){
    var k = list.getAttribute('data-store');
    var st = store.get('chk:' + k, {});
    var boxes = $$('input[type=checkbox]', list);
    var counter = $('[data-count="' + k + '"]');
    function upd(){
      if(!counter) return;
      var n = boxes.filter(function(c){ return c.checked; }).length;
      counter.textContent = n + '/' + boxes.length;
    }
    boxes.forEach(function(c, i){
      c.checked = !!st[i];
      c.addEventListener('change', function(){ st[i] = c.checked; store.set('chk:' + k, st); upd(); });
    });
    upd();
  });

  /* ── prompt builder ── */
  var ids = ['role', 'situ', 'task', 'form'];
  var fields = ids.map(function(k){ return $('#b-' + k); }).filter(Boolean);
  var out = $('#b-out'), meter = $$('.meter i'), bCount = $('#b-count');
  if(fields.length === 4 && out){
    var saved = store.get('builder', null);
    if(saved && saved.length === 4) fields.forEach(function(f, i){ if(typeof saved[i] === 'string') f.value = saved[i]; });
    var compose = function(){
      var vals = fields.map(function(f){ return f.value.trim().replace(/[.。\s]+$/, ''); });
      meter.forEach(function(m, i){ m.classList.toggle('on', !!vals[i]); });
      var filled = vals.filter(Boolean);
      out.textContent = filled.length ? filled.join('. ') + '.' : '네 칸을 채우면 요청 글이 여기에 만들어져요.';
      if(bCount) bCount.textContent = filled.length + ' / 4';
      store.set('builder', fields.map(function(f){ return f.value; }));
    };
    fields.forEach(function(f){ f.addEventListener('input', compose); });
    var reset = $('#b-reset');
    if(reset) reset.addEventListener('click', function(){
      fields.forEach(function(f){ f.value = f.defaultValue; });
      compose(); toast('처음 예시로 돌렸어요');
    });
    compose();
  }

  /* ── collection ── */
  var coll = store.get('coll', []);
  var collList = $('#coll'), collEmpty = $('#coll-empty'), collAll = $('#coll-all');
  function renderColl(){
    if(!collList) return;
    collList.innerHTML = '';
    coll.forEach(function(t, i){
      var li = document.createElement('li');
      var p = document.createElement('p'); p.textContent = t;
      var acts = document.createElement('div'); acts.className = 'acts';
      var c = document.createElement('button'); c.type = 'button'; c.className = 'copy sm'; c.textContent = '복사'; c.setAttribute('data-copy', t);
      var d = document.createElement('button'); d.type = 'button'; d.className = 'ghost'; d.textContent = '빼기';
      d.addEventListener('click', function(){ coll.splice(i, 1); store.set('coll', coll); renderColl(); });
      acts.appendChild(c); acts.appendChild(d);
      li.appendChild(p); li.appendChild(acts);
      collList.appendChild(li);
    });
    if(collEmpty) collEmpty.hidden = coll.length > 0;
    if(collAll) collAll.hidden = coll.length === 0;
    $$('.coll-n').forEach(function(e){ e.textContent = coll.length; });
  }
  document.addEventListener('click', function(e){
    var b = e.target.closest ? e.target.closest('[data-collect]') : null;
    if(!b) return;
    var el = $(b.getAttribute('data-collect'));
    var t = el ? el.innerText.trim() : '';
    if(!t) return;
    if(coll.indexOf(t) === -1){ coll.push(t); store.set('coll', coll); renderColl(); flash(b, true, '담았어요'); toast('나만의 요청 글 모음에 담았어요'); }
    else { flash(b, true, '이미 있어요'); }
  });
  if(collAll) collAll.addEventListener('click', function(){
    copyText(coll.join('\n\n')).then(function(ok){ flash(collAll, ok); if(ok) toast(coll.length + '개를 한꺼번에 복사했어요'); });
  });
  renderColl();

  /* ── reveal ── */
  var rvs = $$('.rv');
  if('IntersectionObserver' in window && !reduce){
    var io = new IntersectionObserver(function(es){
      es.forEach(function(en){ if(en.isIntersecting){ en.target.classList.add('in'); io.unobserve(en.target); } });
    }, { rootMargin: '0px 0px -8% 0px', threshold: 0.06 });
    rvs.forEach(function(el){ io.observe(el); });
  } else {
    rvs.forEach(function(el){ el.classList.add('in'); });
  }

  /* ── nav active + progress ── */
  var nav = $('.nav');
  var links = $$('.nav a');
  var secs = links.map(function(a){ return $(a.getAttribute('href')); });
  function setActive(id){
    links.forEach(function(a){
      var on = a.getAttribute('href') === '#' + id;
      a.classList.toggle('on', on);
      if(on && nav && nav.scrollWidth > nav.clientWidth){
        nav.scrollTo({ left: a.offsetLeft - nav.clientWidth / 2 + a.clientWidth / 2, behavior: reduce ? 'auto' : 'smooth' });
      }
    });
  }
  if('IntersectionObserver' in window){
    var io2 = new IntersectionObserver(function(es){
      es.forEach(function(en){ if(en.isIntersecting) setActive(en.target.id); });
    }, { rootMargin: '-45% 0px -50% 0px' });
    secs.forEach(function(s){ if(s) io2.observe(s); });
  }
  var bar = $('.progress span'), ticking = false;
  function onScroll(){
    if(ticking) return; ticking = true;
    requestAnimationFrame(function(){
      var h = document.documentElement.scrollHeight - innerHeight;
      if(bar) bar.style.width = (h > 0 ? (scrollY / h * 100) : 0) + '%';
      if(scrollY < innerHeight * .5) links.forEach(function(a){ a.classList.remove('on'); });
      ticking = false;
    });
  }
  addEventListener('scroll', onScroll, { passive: true });
  onScroll();
})();
